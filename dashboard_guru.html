// ============================================================ // GOOGLE APPS SCRIPT - KONEKTOR UNTUK SISTEM MANAJEMEN SEKOLAH // ============================================================ // ========== KONFIGURASI ========== // Ganti dengan ID Google Sheet Anda const SPREADSHEET_ID = 'MASUKKAN (paste) ID SPREEDSHEETMU'; // ========== FUNGSI UTAMA ========== function doGet(e) { try { const action = (e && e.parameter) ? e.parameter.action : ''; if (action === 'ping') { return ContentService .createTextOutput(JSON.stringify({ status: 'success', message: 'Koneksi ke Google Sheets berhasil!', timestamp: new Date().toISOString() })) .setMimeType(ContentService.MimeType.JSON); } if (action === 'get_data') { return handleGetData(e); } return ContentService .createTextOutput(JSON.stringify({ status: 'success', message: 'Server siap menerima data POST' })) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService .createTextOutput(JSON.stringify({ status: 'error', message: 'Error: ' + err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } function doPost(e) { try { console.log('📥 Menerima request POST'); if (!e || !e.postData) { console.log('❌ Tidak ada postData'); return ContentService .createTextOutput(JSON.stringify({ status: 'error', message: 'Tidak ada data yang diterima' })) .setMimeType(ContentService.MimeType.JSON); } console.log('📥 Raw data length:', e.postData.contents.length); const data = JSON.parse(e.postData.contents); const action = data.action || ''; const payload = data.data || []; console.log(`📥 Action: ${action}, Jumlah data: ${payload.length}`); // ========== TAMBAHKAN INI UNTUK TARIK DATA ========== if (action === 'get_data') { const result = handleGetDataPost(data); return ContentService .createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } // ========== HANDLE SEMUA ACTION ========== if (action === 'sync_users' || action === 'sync_siswa' || action === 'sync_guru') { return syncUsers(payload); } if (action === 'sync_mapel') { return syncMapel(payload); } if (action === 'sync_absensi') { return syncAbsensi(payload); } if (action === 'sync_nilai') { return syncNilai(payload); } if (action === 'sync_pengumuman') { return syncPengumuman(payload); } if (action === 'sync_jurnal') { return syncJurnal(payload); } return ContentService .createTextOutput(JSON.stringify({ status: 'error', message: 'Action tidak dikenal: ' + action })) .setMimeType(ContentService.MimeType.JSON); } catch(err) { console.log('❌ Error doPost:', err.toString()); return ContentService .createTextOutput(JSON.stringify({ status: 'error', message: 'Error: ' + err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } // ============================================================ // ========== FUNGSI GET DATA VIA POST (TARIK DATA) ========== // ============================================================ function handleGetDataPost(data) { try { const type = data.type || 'all'; const ss = SpreadsheetApp.openById(SPREADSHEET_ID); const result = {}; // Fungsi untuk membaca sheet function getSheetData(sheetName) { const sheet = ss.getSheetByName(sheetName); if (!sheet) return []; const data = sheet.getDataRange().getValues(); if (data.length < 2) return []; const headers = data[0]; const rows = data.slice(1); return rows.map(row => { const obj = {}; headers.forEach((h, i) => { obj[h] = row[i] || ''; }); return obj; }); } if (type === 'all' || type === 'users') { result.users = getSheetData('Users'); } if (type === 'all' || type === 'mapel') { result.mapel = getSheetData('Mapel'); } if (type === 'all' || type === 'absensi') { result.absensi = getSheetData('Absensi'); } if (type === 'all' || type === 'nilai') { result.nilai = getSheetData('Nilai'); } if (type === 'all' || type === 'jurnal') { result.jurnal = getSheetData('Jurnal'); } if (type === 'all' || type === 'pengumuman') { result.pengumuman = getSheetData('Pengumuman'); } return { status: 'success', message: 'Data berhasil diambil', data: result, timestamp: new Date().toISOString() }; } catch(err) { return { status: 'error', message: 'Error: ' + err.toString() }; } } // ============================================================ // ========== FUNGSI GET DATA (TARIK DATA) VIA GET ========== // ============================================================ function handleGetData(e) { try { const type = e.parameter.type || 'all'; const ss = SpreadsheetApp.openById(SPREADSHEET_ID); const result = {}; // Fungsi untuk membaca sheet function getSheetData(sheetName) { const sheet = ss.getSheetByName(sheetName); if (!sheet) return []; const data = sheet.getDataRange().getValues(); if (data.length < 2) return []; const headers = data[0]; const rows = data.slice(1); return rows.map(row => { const obj = {}; headers.forEach((h, i) => { obj[h] = row[i] || ''; }); return obj; }); } if (type === 'all' || type === 'users') { result.users = getSheetData('Users'); } if (type === 'all' || type === 'mapel') { result.mapel = getSheetData('Mapel'); } if (type === 'all' || type === 'absensi') { result.absensi = getSheetData('Absensi'); } if (type === 'all' || type === 'nilai') { result.nilai = getSheetData('Nilai'); } if (type === 'all' || type === 'jurnal') { result.jurnal = getSheetData('Jurnal'); } if (type === 'all' || type === 'pengumuman') { result.pengumuman = getSheetData('Pengumuman'); } return ContentService .createTextOutput(JSON.stringify({ status: 'success', message: 'Data berhasil diambil', data: result, timestamp: new Date().toISOString() })) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService .createTextOutput(JSON.stringify({ status: 'error', message: 'Error: ' + err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } // ============================================================ // ========== FUNGSI SYNC USERS ========== // ============================================================ function syncUsers(data) { try { console.log('📝 Menyimpan data Users...');
