Panduan Integrasi Google Sheets & Apps Script

Langkah mudah menghubungkan portal website ke Google Sheets sekolah Anda

1 Google Sheets

Buat Spreadsheet Baru

Buka sheets.new di browser Anda. Beri nama file, misalnya Database SDN 3 Ngantru. Buat tab bernama Users.

2 Spreadsheet ID

Dapatkan Spreadsheet ID

Lihat alamat URL browser Google Sheet Anda. Salin deretan kode unik di antara /d/ dan /edit. Tempelkan kode ini di Tab 3.

3 Apps Script Editor

Buka Google Apps Script

Di Google Sheets, klik menu top-bar: Ekstensi (Extensions) → Apps Script. Hapus seluruh isi kode bawaan yang ada di editor tersebut.

4 Paste & Save

Tempelkan Kode Script

Buka Tab 2 (Salin Kode Apps Script) di halaman ini, klik tombol salin. Tempelkan kode ke editor Apps Script. Pastikan SPREADSHEET_ID telah diganti. Klik ikon Simpan.

5 Deploy Web App

Terapkan Sebagai Aplikasi Web (Deploy)

A. Klik Terapkan Pilih Terapkan (Deploy) → Penerapan baru (New deployment).
B. Pilih Jenis Web App Klik ikon gerigi → Pilih Aplikasi Web (Web app).
C. Izin Akses (Wajib!) Pada bagian Who has access (Siapa yang memiliki akses), pilih: Siapa saja (Anyone).
6 Testing & Sinkronisasi

Uji Koneksi & Sinkronisasi Users

1. Simpan Kredensial Di Tab 3, isi Spreadsheet ID dan URL Web App, lalu klik Simpan.
2. Uji Koneksi Klik tombol Tes di Tab 3 untuk memastikan koneksi ke Spreadsheet berhasil.
3. Buat Users Buka Tab 4, klik Buat Default Users untuk mengisi data ke Google Sheet.

function initializeSheets() {
    try {
        const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
        let created = [];
        let messages = [];
        
        // ========== BUAT SHEET "Buat_Kuis" dengan struktur Excel ==========
        let sheetBuatKuis = ss.getSheetByName(SHEET_BUAT_KUIS);
        if (!sheetBuatKuis) {
            sheetBuatKuis = ss.insertSheet(SHEET_BUAT_KUIS);
            
            sheetBuatKuis.getRange(1, 1, 1, HEADER_BUAT_KUIS.length).setValues([HEADER_BUAT_KUIS]);
            sheetBuatKuis.getRange(1, 1, 1, HEADER_BUAT_KUIS.length).setFontWeight("bold");
            sheetBuatKuis.setFrozenRows(1);
            
            const lebarKolom = [180, 180, 120, 120, 350, 200, 200, 200, 200, 180];
            for (let i = 0; i < lebarKolom.length; i++) {
                sheetBuatKuis.setColumnWidth(i + 1, lebarKolom[i]);
            }
            
            created.push(SHEET_BUAT_KUIS);
            messages.push(`✅ Sheet "${SHEET_BUAT_KUIS}" berhasil dibuat (struktur Excel)`);
            console.log(`✅ Sheet "${SHEET_BUAT_KUIS}" berhasil dibuat`);
        } else {
            const existingHeader = sheetBuatKuis.getRange(1, 1, 1, HEADER_BUAT_KUIS.length).getValues()[0];
            let needUpdate = false;
            for (let i = 0; i < HEADER_BUAT_KUIS.length; i++) {
                if (existingHeader[i] !== HEADER_BUAT_KUIS[i]) {
                    needUpdate = true;
                    break;
                }
            }
            if (needUpdate) {
                sheetBuatKuis.getRange(1, 1, 1, HEADER_BUAT_KUIS.length).setValues([HEADER_BUAT_KUIS]);
                sheetBuatKuis.getRange(1, 1, 1, HEADER_BUAT_KUIS.length).setFontWeight("bold");
                messages.push(`🔄 Sheet "${SHEET_BUAT_KUIS}" diupdate dengan header baru`);
            }
            messages.push(`ℹ️ Sheet "${SHEET_BUAT_KUIS}" sudah ada`);
            console.log(`ℹ️ Sheet "${SHEET_BUAT_KUIS}" sudah ada`);
        }
        
        // ========== BUAT SHEET "Hasil_Kuis" ==========
        let sheetHasilKuis = ss.getSheetByName(SHEET_HASIL_KUIS);
        if (!sheetHasilKuis) {
            sheetHasilKuis = ss.insertSheet(SHEET_HASIL_KUIS);
            sheetHasilKuis.getRange(1, 1, 1, HEADER_HASIL_KUIS.length).setValues([HEADER_HASIL_KUIS]);
            sheetHasilKuis.getRange(1, 1, 1, HEADER_HASIL_KUIS.length).setFontWeight("bold");
            sheetHasilKuis.setFrozenRows(1);
            
            const lebarKolomHasil = [60, 120, 120, 150, 120, 100, 80, 100, 120, 120, 80, 150, 150, 100, 180];
            for (let i = 0; i < lebarKolomHasil.length; i++) {
                sheetHasilKuis.setColumnWidth(i + 1, lebarKolomHasil[i]);
            }
            
            created.push(SHEET_HASIL_KUIS);
            messages.push(`✅ Sheet "${SHEET_HASIL_KUIS}" berhasil dibuat`);
            console.log(`✅ Sheet "${SHEET_HASIL_KUIS}" berhasil dibuat`);
        } else {
            messages.push(`ℹ️ Sheet "${SHEET_HASIL_KUIS}" sudah ada`);
            console.log(`ℹ️ Sheet "${SHEET_HASIL_KUIS}" sudah ada`);
        }
        
        // ========== BUAT SHEET "Users" ==========
        let sheetUsers = ss.getSheetByName(SHEET_USERS);
        if (!sheetUsers) {
            sheetUsers = ss.insertSheet(SHEET_USERS);
            sheetUsers.getRange(1, 1, 1, HEADER_USERS.length).setValues([HEADER_USERS]);
            sheetUsers.getRange(1, 1, 1, HEADER_USERS.length).setFontWeight("bold");
            sheetUsers.setFrozenRows(1);
            
            const lebarKolomUsers = [120, 150, 150, 200, 350, 150];
            for (let i = 0; i < lebarKolomUsers.length; i++) {
                sheetUsers.setColumnWidth(i + 1, lebarKolomUsers[i]);
            }
            
            created.push(SHEET_USERS);
            messages.push(`✅ Sheet "${SHEET_USERS}" berhasil dibuat`);
            console.log(`✅ Sheet "${SHEET_USERS}" berhasil dibuat`);
        } else {
            const existingHeader = sheetUsers.getRange(1, 1, 1, HEADER_USERS.length).getValues()[0];
            let needUpdate = false;
            for (let i = 0; i < HEADER_USERS.length; i++) {
                if (existingHeader[i] !== HEADER_USERS[i]) {
                    needUpdate = true;
                    break;
                }
            }
            if (needUpdate) {
                sheetUsers.getRange(1, 1, 1, HEADER_USERS.length).setValues([HEADER_USERS]);
                sheetUsers.getRange(1, 1, 1, HEADER_USERS.length).setFontWeight("bold");
                messages.push(`🔄 Sheet "${SHEET_USERS}" diupdate dengan header baru`);

            messages.push(`ℹ️ Sheet "${SHEET_USERS}" sudah ada`);
            console.log(`ℹ️ Sheet "${SHEET_USERS}" sudah ada`);
        }
        
        return {
            status: 'success',
            message: 'Inisialisasi sheet selesai',
            messages: messages,
            created: created,
            timestamp: new Date().toISOString()
        };
        
    } catch(err) {
        console.error('❌ Error initializeSheets:', err.toString());
        return {
            status: 'error',
            message: 'Error: ' + err.toString()
        };
    }
}

function doGet(e) {
    try {
        const action = (e && e.parameter) ? e.parameter.action : '';
        
        if (action === 'init_sheets') {
            const result = initializeSheets();
            return ContentService
                .createTextOutput(JSON.stringify(result))
                .setMimeType(ContentService.MimeType.JSON);
        }
        
        if (action === 'ping') {
            return ContentService
                .createTextOutput(JSON.stringify({
                    status: 'success',
                    message: 'Koneksi ke Google Sheets berhasil!',
                    timestamp: new Date().toISOString()
                }))
                .setMimeType(ContentService.MimeType.JSON);
        }
        
        if (action === 'get_kuis') {
            return handleGetKuis(e);
        }
        
        if (action === 'get_data') {
            return handleGetData(e);
        }
        
        if (action === 'get_buat_kuis') {
            return handleGetBuatKuis(e);
        }
        
        if (action === 'get_hasil_kuis') {
            return handleGetHasilKuis(e);
        }
        
        if (action === 'get_users') {
            return handleGetUsers(e);
        }
        
        return ContentService
            .createTextOutput(JSON.stringify({
                status: 'success',
                message: 'Server siap menerima data POST',
                endpoints: [
                    '?action=ping - Cek koneksi',
                    '?action=init_sheets - Inisialisasi sheet',
                    '?action=get_kuis - Ambil semua kuis (format Excel)',
                    '?action=get_data&type=all - Ambil semua data',
                    '?action=get_hasil_kuis - Ambil semua hasil kuis',
                    '?action=get_users - Ambil semua data users'
                ]
            }))
            .setMimeType(ContentService.MimeType.JSON);
        
    } catch(err) {
        return ContentService
            .createTextOutput(JSON.stringify({
                status: 'error',
                message: 'Error: ' + err.toString()
            }))
            .setMimeType(ContentService.MimeType.JSON);
    }
}

function doPost(e) {
    try {
        console.log('📥 Menerima request POST');
        
        if (!e || !e.postData) {
            console.log('❌ Tidak ada postData');
            return ContentService
                .createTextOutput(JSON.stringify({
                    status: 'error',
                    message: 'Tidak ada data yang diterima'
                }))
                .setMimeType(ContentService.MimeType.JSON);
        }
        
        console.log('📥 Raw data length:', e.postData.contents.length);
        
        const data = JSON.parse(e.postData.contents);
        const action = data.action || '';
        const payload = data.data || [];
        const webAppUrl = data.webAppUrl || '';
        
        console.log(`📥 Action: ${action}, Jumlah data: ${payload.length}`);
        console.log(`📥 WebAppUrl: ${webAppUrl}`);
        
        if (action === 'get_data') {
            const result = handleGetDataPost(data);
            return ContentService
                .createTextOutput(JSON.stringify(result))
                .setMimeType(ContentService.MimeType.JSON);
        }
        
        if (action === 'sync_users' || action === 'sync_siswa' || action === 'sync_guru') {
            return syncUsers(payload, webAppUrl);
        }
        
        if (action === 'sync_mapel') {
            return syncMapel(payload);
        }
        
        if (action === 'sync_absensi') {
            return syncAbsensi(payload);
        }
        
        if (action === 'sync_nilai') {
            return syncNilai(payload);
        }
        
        if (action === 'sync_pengumuman') {
            return syncPengumuman(payload);
        }
        
        if (action === 'sync_jurnal') {
            return syncJurnal(payload);
        }
        
        if (action === 'save_kuis' || action === 'sync_kuis') {
            return handleSaveKuis(data);
        }
        
        if (action === 'import_kuis') {
            return handleImportKuis(data);
        }
        
        if (action === 'save_buat_kuis') {
            return handleSaveBuatKuisLegacy(data);
        }
        
        if (action === 'save_hasil_kuis' || action === 'sync_hasil_kuis') {
            return handleSaveHasilKuis(data);
        }
        
        if (action === 'update_status_kuis') {
            return handleUpdateStatusKuis(data);
        }
        
        if (action === 'delete_kuis') {
            return handleDeleteKuis(data);
        }
        
        if (action === 'clear_all_kuis') {
            return handleClearAllKuis();
        }
        
        return ContentService
            .createTextOutput(JSON.stringify({
                status: 'error',
                message: 'Action tidak dikenal: ' + action
            }))
            .setMimeType(ContentService.MimeType.JSON);
        
    } catch(err) {
        console.log('❌ Error doPost:', err.toString());
        return ContentService
            .createTextOutput(JSON.stringify({
                status: 'error',
                message: 'Error: ' + err.toString()
            }))
            .setMimeType(ContentService.MimeType.JSON);
    }
}

function handleGetUsers(e) {
    try {
        const sheet = getSheetUsers();
        const lastRow = sheet.getLastRow();
        
        if (lastRow < 2) {
            return createResponse(true, "Belum ada data users", { data: [] });
        }
        
        const range = sheet.getRange(2, 1, lastRow - 1, HEADER_USERS.length);
        const values = range.getValues();
        
        const data = values.map(row => {
            let obj = {};
            HEADER_USERS.forEach((header, idx) => {
                obj[header] = row[idx] || '';
            });
            return obj;
        });
        
        return createResponse(true, "Data users berhasil diambil", { 
            data: data, 
            total: data.length 
        });
        
    } catch(error) {
        return createResponse(false, "Gagal mengambil data users: " + error.toString());
    }
}

// ============================================================
function handleSaveKuis(data) {
    try {
        console.log('📝 Menyimpan data kuis ke sheet (format Excel)...');
        console.log('📝 Data yang diterima:', JSON.stringify(data));
        
        const sheet = getSheetBuatKuis();
        const questions = data.questions || [];
        const metadata = data.metadata || {};
        
        if (questions.length === 0) {
            return createResponse(false, "Tidak ada soal untuk disimpan");
        }

        const waktuMulai = metadata.waktu_mulai || new Date().toLocaleString('id-ID');
        const waktuSelesai = metadata.waktu_selesai || new Date(Date.now() + 7*24*60*60*1000).toLocaleString('id-ID');
        const durasi = metadata.durasi || 30;

        console.log(`📝 Waktu Mulai: ${waktuMulai}`);
        console.log(`📝 Waktu Selesai: ${waktuSelesai}`);
        console.log(`📝 Durasi: ${durasi}`);
        console.log(`📝 Jumlah soal: ${questions.length}`);

        const rows = questions.map((q, index) => {
            let jawaban = q.jawaban || '';
            
            if (q.tipe === 'MENJODOHKAN' && q.jodohIndex) {
                jawaban = q.jodohIndex.join(';');
            }
            
            if (q.tipe === 'ISIAN' && q.jawaban_alternatif) {
                jawaban = q.jawaban + ' | ' + q.jawaban_alternatif;
            }

            const row = [
                waktuMulai,
                waktuSelesai,
                durasi,
                q.tipe || 'PG',
                q.pertanyaan || '',
                q.pilihanA || '',
                q.pilihanB || '',
                q.pilihanC || '',
                q.pilihanD || '',
                jawaban
            ];
            
            console.log(`📝 Soal ${index + 1}:`, JSON.stringify(row));
            return row;
        });

        if (rows.length > 0) {
            const lastRow = sheet.getLastRow();
            console.log(`📝 LastRow sebelum simpan: ${lastRow}`);
            
            if (lastRow >= 2) {
                const lastRowData = sheet.getRange(lastRow, 1, 1, 10).getValues()[0];
                const isEmpty = lastRowData.every(cell => cell === '');
                if (!isEmpty) {
                    sheet.appendRow(['', '', '', '', '', '', '', '', '', '']);
                    console.log('📝 Menambahkan baris kosong sebagai pemisah');
                }
            }
            
            let successCount = 0;
            for (let i = 0; i < rows.length; i++) {
                try {
                    sheet.appendRow(rows[i]);
                    successCount++;
                } catch(rowError) {
                    console.error(`❌ Gagal menyimpan baris ${i+1}:`, rowError);
                }
            }
            
            console.log(`✅ ${successCount} dari ${questions.length} soal berhasil disimpan`);
            
            const newLastRow = sheet.getLastRow();
            console.log(`📝 LastRow setelah simpan: ${newLastRow}`);
        }

        return createResponse(true, `Berhasil menyimpan ${questions.length} soal kuis`, {
            total: questions.length,
            timestamp: new Date().toISOString()
        });
        
    } catch(error) {
        console.error("❌ Error save kuis:", error);
        return createResponse(false, "Terjadi kesalahan: " + error.toString());
    }
}

        const gayaBahasa = data.Gaya_Bahasa || data.gaya_bahasa || data.style || "Formal & Edukatif";
        const status = data.Status || data.status || "Aktif";
        
        const questions = data.questions || [];
        const metadata = data.metadata || {};
        
        if (questions.length > 0) {
            console.log(`📝 Menyimpan ${questions.length} soal dengan format Excel...`);
            return handleSaveKuis(data);
        }
        
        const waktuMulai = metadata.waktu_mulai || new Date().toLocaleString('id-ID');
        const waktuSelesai = metadata.waktu_selesai || new Date(Date.now() + 7*24*60*60*1000).toLocaleString('id-ID');
        const durasiNum = parseInt(durasi) || 30;
        
        const row = [
            waktuMulai,
            waktuSelesai,
            durasiNum,
            tipeSoal,
            `📚 ${judul} | Mapel: ${mapel} | Kelas: ${kelas} | Gaya: ${gayaBahasa} | Status: ${status}`,
            '', '', '', '',
            `${jumlahSoal} soal`
        ];
        
        console.log('📝 Row yang akan disimpan:', JSON.stringify(row));
        
        sheet.appendRow(row);
        console.log(`✅ Data kuis "${judul}" berhasil disimpan (legacy) di baris ${sheet.getLastRow()}`);
        
        return createResponse(true, "Data kuis berhasil disimpan", {
            judul: judul,
            row: sheet.getLastRow()
        });
        
    } catch(error) {
        console.error("❌ Error save buat kuis legacy:", error);
        return createResponse(false, "Terjadi kesalahan: " + error.toString());
    }

function handleUpdateStatusKuis(data) {
    try {
        const idKuis = data.ID_Kuis || data.id_kuis;
        const statusBaru = data.Status || data.status;
        
        if (!idKuis || !statusBaru) {
            return createResponse(false, "ID_Kuis dan Status baru wajib diisi");
        }
        
        const sheet = getSheetBuatKuis();
        const dataRange = sheet.getDataRange();
        const values = dataRange.getValues();
        
        for (let i = 1; i < values.length; i++) {
            const pertanyaan = values[i][4] || '';
            if (pertanyaan.includes(idKuis)) {
                const newPertanyaan = pertanyaan.replace(/Status: [A-Za-z]+/, `Status: ${statusBaru}`);
                sheet.getRange(i + 1, 5).setValue(newPertanyaan);
                
                return createResponse(true, "Status kuis berhasil diperbarui", {
                    id_kuis: idKuis,
                    status: statusBaru,
                    row: i + 1
                });
            }
        }
        
        return createResponse(false, "Kuis dengan ID " + idKuis + " tidak ditemukan");
        
    } catch(error) {
        console.error("❌ Error update status:", error);
        return createResponse(false, "Terjadi kesalahan: " + error.toString());
    }
}

function handleGetHasilKuis(e) {
    try {
        const sheet = getSheetHasilKuis();

        if (lastRow < 2) {
            return createResponse(true, "Belum ada data hasil kuis", { data: [] });
        }
        
        const range = sheet.getRange(2, 1, lastRow - 1, HEADER_HASIL_KUIS.length);
        const values = range.getValues();
        
        const data = values.map(row => {
            let obj = {};
            HEADER_HASIL_KUIS.forEach((header, idx) => {
                obj[header] = row[idx] || '';
            });
            return obj;
        });
        
        if (e && e.parameter && e.parameter.id_kuis) {
            const idKuis = e.parameter.id_kuis;
            const filtered = data.filter(item => item.ID_Kuis === idKuis);
            return createResponse(true, "Data hasil kuis berhasil diambil", { 
                data: filtered, 
                total: filtered.length,
                id_kuis: idKuis
            });
        }
        
        return createResponse(true, "Data hasil kuis berhasil diambil", { 
            data: data, 
            total: data.length 
        });
        
    } catch(error) {
        return createResponse(false, "Gagal mengambil data hasil kuis: " + error.toString());
    }
}

// ========== 🚀 FUNGSI SYNC USERS (FIXED - PRIORITAS DATA DARI CLIENT) ==========
// ============================================================
function syncUsers(data, webAppUrl) {
    try {
        console.log('📝 Menyimpan data Users...');
        console.log('📝 Data length:', data ? data.length : 0);
        console.log('📝 WebAppUrl:', webAppUrl);
        
        const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
        let sheet = ss.getSheetByName(SHEET_USERS);
        if (!sheet) {
            console.log('📝 Membuat sheet Users...');
            sheet = ss.insertSheet(SHEET_USERS);
            sheet.getRange(1, 1, 1, HEADER_USERS.length).setValues([HEADER_USERS]);
            sheet.getRange(1, 1, 1, HEADER_USERS.length).setFontWeight('bold');
            sheet.setFrozenRows(1);
        }
        
        // ========== 1. BACA DATA EXISTING DARI SHEET ==========
        const existingData = sheet.getDataRange().getValues();
        const headerRow = existingData[0];
        
        const roleIdx = headerRow.indexOf('Role');
        const usernameIdx = headerRow.indexOf('Username');
        const passwordIdx = headerRow.indexOf('Password');
        const namaIdx = headerRow.indexOf('Nama');
        const urlIdx = headerRow.indexOf('URL_Script_App');
        const kelasIdx = headerRow.indexOf('Kelas');
        
        let adminData = null;
        let guruData = null;
        let existingSiswa = [];
        let existingUsernames = new Set();
        
        for (let i = 1; i < existingData.length; i++) {
            const row = existingData[i];
            if (row.length < 2 || !row[0]) continue;
            
            const role = String(row[roleIdx] || '').toLowerCase().trim();
            const username = String(row[usernameIdx] || '').trim();
            
            if (!username) continue;
            existingUsernames.add(username);
            
            if (role === 'admin') {
                adminData = {
                    Role: 'Admin',
                    Username: username,
                    Password: String(row[passwordIdx] || '123'),
                    Nama: String(row[namaIdx] || 'Administrator'),
                    URL_Script_App: String(row[urlIdx] || ''),
                    Kelas: String(row[kelasIdx] || 'System')
                };
                console.log(`🔄 Mempertahankan Admin: ${username}`);
            } else if (role === 'guru') {
                guruData = {
                    Role: 'Guru',
                    Username: username,
                    Password: String(row[passwordIdx] || '123'),
                    Nama: String(row[namaIdx] || 'Guru Pengajar'),
                    URL_Script_App: String(row[urlIdx] || ''),
                    Kelas: String(row[kelasIdx] || '')
                };
                console.log(`🔄 Mempertahankan Guru: ${username}`);
            } else if (role === 'siswa') {
                existingSiswa.push({
                    Role: 'Siswa',
                    Username: username,
                    Password: String(row[passwordIdx] || '123'),
                    Nama: String(row[namaIdx] || ''),
                    URL_Script_App: String(row[urlIdx] || ''),
                    Kelas: String(row[kelasIdx] || 'Peserta Didik')
                });
                console.log(`📝 Siswa existing: ${username}`);
            }
        }
        
        console.log(`🔄 Admin ditemukan: ${adminData ? adminData.Username : 'TIDAK ADA'}`);
        console.log(`🔄 Guru ditemukan: ${guruData ? guruData.Username : 'TIDAK ADA'}`);
        console.log(`📝 ${existingSiswa.length} siswa existing`);
        
        // ========== 2. PROSES DATA DARI CLIENT ==========
        let clientAdmin = null;
        let clientGuru = null;
        let clientSiswa = [];
        
        if (data && data.length > 0) {
            for (let i = 0; i < data.length; i++) {
                const item = data[i];
                const role = String(item.Role || '').toLowerCase().trim();
                const username = String(item.Username || '').trim();
                const nama = String(item.Nama || '').trim();
                
                if (!username) continue;
                
                if (role === 'admin') {
                    clientAdmin = {
                        Role: 'Admin',
                        Username: username,
                        Password: String(item.Password || '123'),
                        Nama: nama || 'Administrator',
                        URL_Script_App: webAppUrl || String(item.URL_Script_App || ''),
                        Kelas: String(item.Kelas || 'System')
                    };
                    console.log(`📤 Admin dari client: ${username} - ${nama}`);
                } else if (role === 'guru') {
                    clientGuru = {
                        Role: 'Guru',
                        Username: username,
                        Password: String(item.Password || '123'),
                        Nama: nama || 'Guru Pengajar',
                        URL_Script_App: webAppUrl || String(item.URL_Script_App || ''),
                        Kelas: String(item.Kelas || '')
                    };
                    console.log(`📤 Guru dari client: ${username} - ${nama}`);
                } else if (role === 'siswa') {
                    // Cek apakah siswa sudah ada
                    if (!existingUsernames.has(username)) {
                        clientSiswa.push({
                            Role: 'Siswa',
                            Username: username,
                            Password: String(item.Password || '123'),
                            Nama: nama || '',
                            URL_Script_App: webAppUrl || String(item.URL_Script_App || ''),
                            Kelas: String(item.Kelas || 'Peserta Didik')
                        });
                        console.log(`➕ Siswa baru dari client: ${username} - ${nama}`);
                    } else {
                        console.log(`⚠️ Siswa ${username} sudah ada, skip`);
                    }
                }
            }
        }
        
        // ========== 3. PRIORITAS: DATA DARI CLIENT LEBIH UTAMA ==========
        // ADMIN: client > existing > default
        let finalAdmin = null;
        if (clientAdmin) {
            finalAdmin = clientAdmin;
            console.log(`✅ ADMIN menggunakan data dari client: ${finalAdmin.Username}`);
        } else if (adminData) {
            finalAdmin = adminData;
            // Tapi URL_Script_App tetap pakai webAppUrl dari client
            finalAdmin.URL_Script_App = webAppUrl || finalAdmin.URL_Script_App;
            console.log(`✅ ADMIN menggunakan data existing: ${finalAdmin.Username}`);
        } else {
            finalAdmin = {
                Role: 'Admin',
                Username: 'admin',
                Password: '123',
                Nama: 'Administrator',
                URL_Script_App: webAppUrl || '',
                Kelas: 'System'
            };
            console.log('➕ ADMIN default dibuat');
        }
        
        // GURU: client > existing > default
        let finalGuru = null;
        if (clientGuru) {
            finalGuru = clientGuru;
            console.log(`✅ GURU menggunakan data dari client: ${finalGuru.Username}`);
        } else if (guruData) {
            finalGuru = guruData;
            // Tapi URL_Script_App tetap pakai webAppUrl dari client
            finalGuru.URL_Script_App = webAppUrl || finalGuru.URL_Script_App;
            console.log(`✅ GURU menggunakan data existing: ${finalGuru.Username}`);
        } else {
            finalGuru = {
                Role: 'Guru',
                Username: 'guru',
                Password: '123',
                Nama: 'Guru Pengajar',
                URL_Script_App: webAppUrl || '',
                Kelas: ''
            };
            console.log('➕ GURU default dibuat');
        }
        
        console.log(`📝 Menambahkan ${clientSiswa.length} siswa baru`);
        
        // ========== 4. GABUNGKAN SEMUA DATA ==========
        let allRows = [];
        
        // Admin
        allRows.push([
            finalAdmin.Role,
            finalAdmin.Username,
            finalAdmin.Password,
            finalAdmin.Nama,
            finalAdmin.URL_Script_App,
            finalAdmin.Kelas
        ]);
        console.log(`✅ Admin: ${finalAdmin.Username} - ${finalAdmin.Nama} | URL: ${finalAdmin.URL_Script_App}`);
        
        // Guru
        allRows.push([
            finalGuru.Role,
            finalGuru.Username,
            finalGuru.Password,
            finalGuru.Nama,
            finalGuru.URL_Script_App,
            finalGuru.Kelas
        ]);
        console.log(`✅ Guru: ${finalGuru.Username} - ${finalGuru.Nama} | URL: ${finalGuru.URL_Script_App}`);
        
        // Siswa existing
        if (existingSiswa.length > 0) {
            for (const s of existingSiswa) {
                // Update URL dengan webAppUrl terbaru
                s.URL_Script_App = webAppUrl || s.URL_Script_App;
                allRows.push([
                    s.Role,
                    s.Username,
                    s.Password,
                    s.Nama,
                    s.URL_Script_App,
                    s.Kelas
                ]);
            }
            console.log(`✅ ${existingSiswa.length} siswa existing dipertahankan`);
        }
        
        // Siswa baru dari client
        if (clientSiswa.length > 0) {
            for (const s of clientSiswa) {
                allRows.push([
                    s.Role,
                    s.Username,
                    s.Password,
                    s.Nama,
                    s.URL_Script_App,
                    s.Kelas
                ]);
            }
            console.log(`✅ ${clientSiswa.length} siswa baru ditambahkan`);
        }
        
        // ========== 5. TULIS KE SHEET ==========
        const lastRow = sheet.getLastRow();
        if (lastRow > 1) {
            sheet.deleteRows(2, lastRow - 1);
            console.log('🗑️ Data lama dihapus');
        }
        
        if (allRows.length > 0) {
            sheet.getRange(2, 1, allRows.length, HEADER_USERS.length).setValues(allRows);
            console.log(`✅ ${allRows.length} baris data Users tersimpan`);
        }
        
        // ========== 6. RESPONSE ==========
        return ContentService
            .createTextOutput(JSON.stringify({
                status: 'success',
                message: `Data Users berhasil disinkronisasi. ${allRows.length} total records.`,
                data: {
                    total: allRows.length,
                    admin: finalAdmin ? 1 : 0,
                    guru: finalGuru ? 1 : 0,
                    siswa_existing: existingSiswa.length,
                    siswa_baru: clientSiswa.length,
                    webAppUrl: webAppUrl
                },
                timestamp: new Date().toISOString()
            }))
            .setMimeType(ContentService.MimeType.JSON);
        
    } catch(err) {
        console.log('❌ Error syncUsers:', err.toString());
        return ContentService
            .createTextOutput(JSON.stringify({
                status: 'error',
                message: 'Error syncUsers: ' + err.toString()
            }))
            .setMimeType(ContentService.MimeType.JSON);
    }
}
