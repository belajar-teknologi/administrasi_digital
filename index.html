Kode Apps Script untuk Google Sheets

Salin kode di bawah ini dan tempelkan di Google Apps Script Editor, lalu deploy sebagai Web App:

// ============================================================ // GOOGLE APPS SCRIPT - KONEKTOR UNTUK SISTEM MANAJEMEN SEKOLAH // ============================================================ // ========== KONFIGURASI ========== // Ganti dengan ID Google Sheet Anda const SPREADSHEET_ID = '1B8aZ1GtIt7WMI37imqq68lGhnACp2yCjszpI3j4nFeg'; // ========== FUNGSI UTAMA ========== function doGet(e) { try { const action = (e && e.parameter) ? e.parameter.action : ''; if (action === 'ping') { return ContentService .createTextOutput(JSON.stringify({ status: 'success', message: 'Koneksi ke Google Sheets berhasil!', timestamp: new Date().toISOString() })) .setMimeType(ContentService.MimeType.JSON); } return ContentService .createTextOutput(JSON.stringify({ status: 'success', message: 'Server siap menerima data POST' })) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService .createTextOutput(JSON.stringify({ status: 'error', message: 'Error: ' + err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } function doPost(e) { try { console.log('📥 Menerima request POST'); if (!e || !e.postData) { console.log('❌ Tidak ada postData'); return ContentService .createTextOutput(JSON.stringify({ status: 'error', message: 'Tidak ada data yang diterima' })) .setMimeType(ContentService.MimeType.JSON); } console.log('📥 Raw data length:', e.postData.contents.length); const data = JSON.parse(e.postData.contents); const action = data.action || ''; const payload = data.data || []; console.log(`📥 Action: ${action}, Jumlah data: ${payload.length}`); if (action === 'sync_users' || action === 'sync_siswa' || action === 'sync_guru') { return syncUsers(payload); } if (action === 'sync_mapel') { return syncMapel(payload); } if (action === 'sync_absensi') { return syncAbsensi(payload); } if (action === 'sync_nilai') { return syncNilai(payload); } if (action === 'sync_pengumuman') { return syncPengumuman(payload); } return ContentService .createTextOutput(JSON.stringify({ status: 'error', message: 'Action tidak dikenal: ' + action })) .setMimeType(ContentService.MimeType.JSON); } catch(err) { console.log('❌ Error doPost:', err.toString()); return ContentService .createTextOutput(JSON.stringify({ status: 'error', message: 'Error: ' + err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } // ========== FUNGSI SYNC USERS ========== function syncUsers(data) { try { console.log('📝 Menyimpan data Users...'); console.log('📝 Data length:', data ? data.length : 0); const ss = SpreadsheetApp.openById(SPREADSHEET_ID); let sheet = ss.getSheetByName('Users'); if (!sheet) { console.log('📝 Membuat sheet Users...'); sheet = ss.insertSheet('Users'); } // Clear existing data sheet.clear(); // Set headers const headers = ['Role', 'Username', 'Password', 'Nama', 'ID_Tambahan', 'Kelas']; sheet.getRange(1, 1, 1, headers.length).setValues([headers]); // Set data if (data && data.length > 0) { const rows = data.map(item => [ item.Role || '', item.Username || '', item.Password || '', item.Nama || '', item.ID_Tambahan || '', item.Kelas || '' ]); if (rows.length > 0) { sheet.getRange(2, 1, rows.length, headers.length).setValues(rows); console.log(`✅ ${rows.length} baris data Users tersimpan`); } } return ContentService .createTextOutput(JSON.stringify({ status: 'success', message: 'Data Users berhasil disinkronisasi (' + (data ? data.length : 0) + ' records)' })) .setMimeType(ContentService.MimeType.JSON); } catch(err) { console.log('❌ Error syncUsers:', err.toString()); return ContentService .createTextOutput(JSON.stringify({ status: 'error', message: 'Error syncUsers: ' + err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } // ========== FUNGSI SYNC MAPEL ========== function syncMapel(data) { try { console.log('📝 Menyimpan data Mapel...'); const ss = SpreadsheetApp.openById(SPREADSHEET_ID); let sheet = ss.getSheetByName('Mapel'); if (!sheet) { sheet = ss.insertSheet('Mapel'); } sheet.clear(); const headers = ['ID', 'Kode', 'Nama', 'KKM']; sheet.getRange(1, 1, 1, headers.length).setValues([headers]); if (data && data.length > 0) { const rows = data.map(item => [ item.ID || '', item.Kode || '', item.Nama || '', item.KKM || 75 ]); if (rows.length > 0) { sheet.getRange(2, 1, rows.length, headers.length).setValues(rows); console.log(`✅ ${rows.length} baris data Mapel tersimpan`); } } return ContentService .createTextOutput(JSON.stringify({ status: 'success', message: 'Data Mapel berhasil disinkronisasi (' + (data ? data.length : 0) + ' records)' })) .setMimeType(ContentService.MimeType.JSON); } catch(err) { console.log('❌ Error syncMapel:', err.toString()); return ContentService .createTextOutput(JSON.stringify({ status: 'error', message: 'Error syncMapel: ' + err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } // ========== FUNGSI SYNC ABSENSI ========== function syncAbsensi(data) { try { console.log('📝 Menyimpan data Absensi...'); const ss = SpreadsheetApp.openById(SPREADSHEET_ID); let sheet = ss.getSheetByName('Absensi'); if (!sheet) { sheet = ss.insertSheet('Absensi'); } sheet.clear(); const headers = ['Tanggal', 'Kelas', 'Username', 'Nama_Siswa', 'Status_Kehadiran']; sheet.getRange(1, 1, 1, headers.length).setValues([headers]); if (data && data.length > 0) { const rows = data.map(item => [ item.Tanggal || '', item.Kelas || '', item.Username || '', item.Nama_Siswa || '', item.Status_Kehadiran || 'Hadir' ]); if (rows.length > 0) { sheet.getRange(2, 1, rows.length, headers.length).setValues(rows); console.log(`✅ ${rows.length} baris data Absensi tersimpan`); } } return ContentService .createTextOutput(JSON.stringify({ status: 'success', message: 'Data Absensi berhasil disinkronisasi (' + (data ? data.length : 0) + ' records)' })) .setMimeType(ContentService.MimeType.JSON); } catch(err) { console.log('❌ Error syncAbsensi:', err.toString()); return ContentService .createTextOutput(JSON.stringify({ status: 'error', message: 'Error syncAbsensi: ' + err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } // ========== FUNGSI SYNC NILAI ========== function syncNilai(data) { try { console.log('📝 Menyimpan data Nilai...'); const ss = SpreadsheetApp.openById(SPREADSHEET_ID); let sheet = ss.getSheetByName('Nilai'); if (!sheet) { sheet = ss.insertSheet('Nilai'); } sheet.clear(); const headers = ['Kelas', 'Mapel', 'Username', 'Nama_Siswa', 'Harian_1', 'Harian_2', 'Harian_3', 'Harian_4', 'Harian_5', 'Harian_6', 'Harian_7', 'Harian_8', 'Harian_9', 'Harian_10', 'UTS', 'UAS', 'Nilai_Akhir']; sheet.getRange(1, 1, 1, headers.length).setValues([headers]); if (data && data.length > 0) { const rows = data.map(item => [ item.Kelas || '', item.Mapel || '', item.Username || '', item.Nama_Siswa || '', item.Harian_1 || 0, item.Harian_2 || 0, item.Harian_3 || 0, item.Harian_4 || 0, item.Harian_5 || 0, item.Harian_6 || 0, item.Harian_7 || 0, item.Harian_8 || 0, item.Harian_9 || 0, item.Harian_10 || 0, item.UTS || 0, item.UAS || 0, item.Nilai_Akhir || 0 ]); if (rows.length > 0) { sheet.getRange(2, 1, rows.length, headers.length).setValues(rows); console.log(`✅ ${rows.length} baris data Nilai tersimpan`); } } return ContentService .createTextOutput(JSON.stringify({ status: 'success', message: 'Data Nilai berhasil disinkronisasi (' + (data ? data.length : 0) + ' records)' })) .setMimeType(ContentService.MimeType.JSON); } catch(err) { console.log('❌ Error syncNilai:', err.toString()); return ContentService .createTextOutput(JSON.stringify({ status: 'error', message: 'Error syncNilai: ' + err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } // ========== FUNGSI SYNC PENGUMUMAN ========== function syncPengumuman(data) { try { console.log('📝 Menyimpan data Pengumuman...'); const ss = SpreadsheetApp.openById(SPREADSHEET_ID); let sheet = ss.getSheetByName('Pengumuman'); if (!sheet) { sheet = ss.insertSheet('Pengumuman'); } sheet.clear(); const headers = ['ID', 'Tanggal', 'Isi', 'Pembuat']; sheet.getRange(1, 1, 1, headers.length).setValues([headers]); if (data && data.length > 0) { const rows = data.map(item => [ item.ID || '', item.Tanggal || '', item.Isi || '', item.Pembuat || 'Admin' ]);
